题目:(2) 设计一个线段类有两个顶点，一个圆类有圆心和半径，一个算法类有一个线段对象和一个圆的对象，能够求圆心到该线段的最短距离，能够判断线段和圆是相切、相离还是相交。
```cpp
#include <iostream>
#include <cmath>
#include <iomanip>
using namespace std;

class Line_thy {
public:
double x1, y1;
double x2, y2;
Line_thy(double a, double b, double c, double d) {x1 = a;y1 = b;x2 = c;y2 = d;}
};

class Circle_thy {
public:
double x, y;
double r;
Circle_thy(double a, double b, double radius) {x = a;y = b;r = radius;}
};

class Algorithm_thy {
public:
    static double shortestDistance(const Line_thy& line,const Circle_thy& circle) {
double dx = line.x2 - line.x1;
double dy = line.y2 - line.y1;
double len2 = dx * dx + dy * dy;
if (len2 == 0) {
    return sqrt((circle.x - line.x1) * (circle.x - line.x1) +
                (circle.y - line.y1) * (circle.y - line.y1));
}
double t = ((circle.x - line.x1) * dx +(circle.y - line.y1) * dy) / len2;
	if (t < 0)
		t = 0;
	else if (t > 1)
		t = 1;
double px = line.x1 + t * dx;
double py = line.y1 + t * dy;
return sqrt((circle.x - px) * (circle.x - px) +(circle.y - py) * (circle.y - py));}
static void judge(const Line_thy& line,const Circle_thy& circle) {
double d = shortestDistance(line, circle);
const double EPS = 1e-9;
cout << fixed << setprecision(2);
cout << "圆心到线段的最短距离：" << d << endl;
if (fabs(d - circle.r) < EPS) {
cout << "线段与圆相切" << endl;
} else if (d > circle.r) {
cout << "线段与圆相离" << endl;
} else {
cout << "线段与圆相交" << endl;
		}
	}
};
int main() {
    double x1, y1, x2, y2;
    double cx, cy, r;
    cout << "请输入线段两个端点的坐标 x1 y1 x2 y2：";
    cin >> x1 >> y1 >> x2 >> y2;
    cout << "请输入圆心坐标和半径 cx cy r：";
    cin >> cx >> cy >> r;
    if (r < 0) {
        cout << "半径不能为负数！" << endl;
        return 1;
}
Line_thy line(x1, y1, x2, y2);
Circle_thy circle(cx, cy, r);
Algorithm_thy::judge(line, circle);
return 0;
}
```
